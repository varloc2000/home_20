### Context & Objectives

Create a .github/workflows/deploy.yml in order to deploy current repo to remote hetzner server
Use following sample as an example of deployment structure:

```
name: Deploy n8n to stg or prod

on:
  push:
    branches:
      - master
      - staging
  workflow_dispatch:
    inputs:
      N8N_UPGRADE:
        description: 'N8N_UPGRADE: Whether to upgrade n8n images (true/false)'
        required: false
        default: 'false'
      target_branch:
        description: 'Branch to deploy (must match master or staging)'
        required: true
        default: 'master'
        type: choice
        options:
          - master
          - staging

jobs:
  set-env:
    runs-on: ubuntu-latest
    outputs:
      env_name: ${{ steps.branch_check.outputs.env_name }}
      arch: ${{ steps.branch_check.outputs.arch }}
    steps:
      - id: branch_check
        run: |
          # Use the input branch if manual, otherwise use the branch that triggered the push
          SELECTED_BRANCH=${{ github.event.inputs.target_branch || github.ref_name }}

          if [ "$SELECTED_BRANCH" = "master" ]; then
            echo "env_name=prod" >> $GITHUB_OUTPUT
            echo "arch=linux/arm64" >> $GITHUB_OUTPUT
          else
            echo "env_name=stg" >> $GITHUB_OUTPUT
            echo "arch=linux/amd64" >> $GITHUB_OUTPUT
          fi

  build:
    needs: set-env
    runs-on: ${{ needs.set-env.outputs.env_name == 'prod' && 'ubuntu-24.04-arm' || 'ubuntu-latest' }}
    environment: ${{ needs.set-env.outputs.env_name }}
    permissions:
      contents: read
      packages: write
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Log in to Docker Registry
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Build and push via Bake Action
        uses: docker/bake-action@v6
        with:
          files: ./docker-compose.prod.yml
          push: true
          set: |
            *.platform=${{ needs.set-env.outputs.arch }}
            *.cache-from=type=gha,scope=${{ needs.set-env.outputs.env_name }}
            *.cache-to=type=gha,mode=max,scope=${{ needs.set-env.outputs.env_name }}
        env:
          CI_TAG: ${{ needs.set-env.outputs.env_name }}

  deploy:
    needs: [set-env, build]
    runs-on: ubuntu-latest
    environment: ${{ needs.set-env.outputs.env_name }}
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Install SSH client
        run: |
          sudo apt-get update
          sudo apt-get install -y openssh-client

      - name: Start SSH agent
        uses: webfactory/ssh-agent@v0.7.0
        with:
          ssh-private-key: ${{ secrets.SSH_PRIVATE_KEY }}

      - name: Create project directory on server
        run: |
          ssh -o StrictHostKeyChecking=no \
          root@${{ secrets.SSH_HOST }} "mkdir -p /srv/n8n"

      - name: Login to registry
        run: |
          ssh -o StrictHostKeyChecking=no root@${{ secrets.SSH_HOST }} "docker login -u ${{ github.actor }} -p ${{ secrets.GITHUB_TOKEN }} ghcr.io"

      - name: Copy docker-compose.yml to remote server
        run: scp -o StrictHostKeyChecking=no ./docker-compose.prod.yml ./Dockerfile.runners root@${{ secrets.SSH_HOST }}:/srv/n8n/

      - name: Deploy to remote server via SSH
        env:
          CI_N8N_BASIC_AUTH_PASSWORD: ${{ secrets.N8N_BASIC_AUTH_PASSWORD }}
          CI_N8N_ENCRYPTION_KEY: ${{ secrets.N8N_ENCRYPTION_KEY }}
          CI_DB_TYPE: ${{ vars.CI_DB_TYPE }}
          CI_DB_POSTGRESDB_DATABASE: ${{ vars.CI_DB_POSTGRESDB_DATABASE }}
          CI_DB_POSTGRESDB_HOST: ${{ vars.CI_DB_POSTGRESDB_HOST }}
          CI_DB_POSTGRESDB_USER: ${{ vars.CI_DB_POSTGRESDB_USER }}
          CI_DB_POSTGRESDB_PASSWORD: ${{ secrets.CI_DB_POSTGRESDB_PASSWORD }}
          CI_DB_POSTGRESDB_SSL_ENABLED: ${{ vars.CI_DB_POSTGRESDB_SSL_ENABLED }}
          CI_N8N_UPGRADE: ${{ github.event.inputs.N8N_UPGRADE || 'false' }}
          CI_N8N_SUBDOMAIN: ${{ vars.CI_N8N_SUBDOMAIN }}
          CI_STG_SUBDOMAIN: ${{ vars.CI_STG_SUBDOMAIN }}
          CI_NODE_FUNCTION_ALLOW_BUILTIN: ${{ vars.CI_NODE_FUNCTION_ALLOW_BUILTIN }}
          CI_NODE_FUNCTION_ALLOW_EXTERNAL: ${{ vars.CI_NODE_FUNCTION_ALLOW_EXTERNAL }}
          CI_N8N_RUNNERS_AUTH_TOKEN: ${{ secrets.CI_N8N_RUNNERS_AUTH_TOKEN }}
          CI_TAG: ${{ needs.set-env.outputs.env_name }}
        run: |
          ssh -o StrictHostKeyChecking=no -o SendEnv=CI_* root@${{ secrets.SSH_HOST }} << 'EOF'
            cd /srv/n8n/
            if [ "$CI_N8N_UPGRADE" = "true" ]; then
              echo "Pulling latest images for n8n..."
              docker compose -f ./docker-compose.prod.yml pull
            else
              echo "Skipping image pull."
            fi
            docker compose -f ./docker-compose.prod.yml up -d --force-recreate --remove-orphans
          EOF

```

Create separate dedicated docker-compose.prod.yml for prod setup.
Prod container will use traefik as proxy.
Use following docker compose content as an example of how containers should be labeled for traefik
All tools used in repo that have web dashboards have to be accessible via public subdomains

```
version: '3.8'

services:
  n8n:
    image: n8nio/n8n:2.34.6
    container_name: n8n
    restart: unless-stopped
    ports:
      - "127.0.0.1:5678:5678"
    environment:
      GENERIC_TIMEZONE: Europe/Warsaw
    volumes:
      - n8n_data:/home/node/.n8n
      - ./local-files:/files
    networks:
      traefik-network:
        aliases:
          - home20.run.place
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.n8n.rule=Host(`home20.run.place`)"
      - "traefik.http.routers.n8n.entrypoints=websecure"
      - "traefik.http.routers.n8n.tls.certresolver=myresolver"
      - "traefik.http.services.n8n.loadbalancer.server.port=5678"
      - "traefik.http.services.n8n.loadbalancer.server.scheme=http"
      - "traefik.docker.network=traefik-network"

  task-runners:
    image: ghcr.io/varloc2000/n8n-runners:${CI_TAG:-latest}
    build:
      context: .
      dockerfile: Dockerfile.runners
    container_name: n8n-runners
    environment:
      N8N_RUNNERS_TASK_BROKER_URI: "http://n8n:5679"
      N8N_RUNNERS_AUTH_TOKEN: ${CI_N8N_RUNNERS_AUTH_TOKEN}
      N8N_RUNNERS_LAUNCHER_LOG_LEVEL: "debug"
    volumes:
      - n8n_data:/home/node/.n8n
      - ./local-files:/files
    depends_on:
      - n8n
    networks:
      - traefik-network

networks:
  traefik-network:
    external: true

volumes:
  n8n_data:
    driver: local
```

Use home20.run.place as main Host for homeassistant. Add subdomains for other dashboards like zigbee2mqtt.home20.run.place.
